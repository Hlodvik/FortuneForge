// Typography-study layout only. Every full source canvas uses one common scale.
import { createRequire } from 'node:module'
import { readFile, writeFile } from 'node:fs/promises'
import { createHash } from 'node:crypto'
import { dirname, resolve } from 'node:path'
import { fileURLToPath } from 'node:url'

const require = createRequire(import.meta.url)
const sharp = require('C:/Users/Lctiv/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/node_modules/sharp')
const root = dirname(fileURLToPath(import.meta.url))
const previousRoot = resolve(root, '../revision-2')
const hash = bytes => createHash('sha256').update(bytes).digest('hex')
const escape = value => String(value).replaceAll('&', '&amp;').replaceAll('<', '&lt;').replaceAll('>', '&gt;').replaceAll('"', '&quot;')
const sourceManifest = JSON.parse(await readFile(resolve(root, 'font-studies-manifest.json'), 'utf8'))
const previous = JSON.parse(await readFile(resolve(previousRoot, 'preview-board-manifest.json'), 'utf8'))
const approved = previous.assets.filter(asset => asset.group === 'accepted' || asset.id === 'triple-sevens')
if (approved.length !== 11 || sourceManifest.families.length !== 3) throw new Error('Expected eleven approved images and three font studies')

const approvedProof = []
for (const asset of approved) {
  const file = resolve(previousRoot, asset.file)
  const currentSha256 = hash(await readFile(file))
  if (currentSha256 !== asset.sha256) throw new Error(`Approved image changed: ${asset.id}`)
  approvedProof.push({ id: asset.id, source: file, sha256: currentSha256, matchesRevision2: true, readOnly: true })
}

const glyphs = []
for (const family of sourceManifest.families) {
  if (family.assets.map(asset => asset.letter).join('') !== 'AKQJ') throw new Error(`Unexpected glyph order: ${family.id}`)
  for (const item of family.assets) {
    const source = await readFile(resolve(root, item.file))
    if (hash(source) !== item.sha256) throw new Error(`Native glyph source changed: ${item.file}`)
    const { data, info } = await sharp(source).ensureAlpha().raw().toBuffer({ resolveWithObject: true })
    if (info.width !== 1254 || info.height !== 1254 || item.baseline !== 974 || family.commonBaseline !== 974) throw new Error('A common full canvas and baseline are required')
    let minAlpha = 255, maxAlpha = 0, visible = 0
    for (let i = 0; i < data.length; i += 4) {
      const alpha = data[i + 3]
      minAlpha = Math.min(minAlpha, alpha); maxAlpha = Math.max(maxAlpha, alpha)
      if (alpha > 0) {
        visible++
        if (data[i] !== 241 || data[i+1] !== 231 || data[i+2] !== 213) throw new Error(`Non-flat RGB colour found: ${item.file}`)
      }
    }
    if (minAlpha !== 0 || maxAlpha !== 255 || !visible) throw new Error(`Invalid alpha or empty glyph: ${item.file}`)
    // Full-canvas resize: never extract, stretch or individually normalize ink.
    const fullCanvasPreview = await sharp(source).resize(300, 300, { fit: 'contain' }).png().toBuffer()
    glyphs.push({ familyId: family.id, familyTitle: family.title, letter: item.letter, file: item.file,
      sourceSha256: item.sha256, width: 1254, height: 1254, baseline: 974,
      fontSource: family.fontSource, fontSourceSha256: family.fontSourceSha256,
      fontSize: family.fontSize, nominalCapHeight: family.nominalCapHeight,
      visibleRGB: [241,231,213], alphaRange: [minAlpha,maxAlpha], visiblePixels: visible,
      preview: fullCanvasPreview })
  }
}
if (glyphs.length !== 12) throw new Error('Expected twelve glyph PNGs')

const width = 1720, height = 1264, background = '#071321'
const rowY = [180, 534, 888], colX = [320, 650, 980, 1310]
const labelLines = { bodoni: ['Bodoni MT', 'Bold'], georgia: ['Georgia', 'Bold'], times: ['Times New', 'Roman Bold'] }
const backdrop = `<svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${height}">
<rect width="100%" height="100%" fill="${background}"/>
<text x="60" y="61" font-family="Arial, sans-serif" font-size="37" font-weight="700" letter-spacing="2" fill="#f1e7d5">MIDNIGHT VAULT · LETTER STUDIES</text>
<text x="60" y="101" font-family="Arial, sans-serif" font-size="24" fill="#acb8c7">Actual typefaces · flat warm ivory · no outline, bevel, shadow or texture</text>
${['A','K','Q','J'].map((letter,column) => `<text x="${colX[column]+150}" y="150" text-anchor="middle" font-family="Arial, sans-serif" font-size="23" fill="#91a2b6">${letter}</text>`).join('')}
${sourceManifest.families.map((family,row) => {
  const lines = labelLines[family.id]
  return `<text x="60" y="${rowY[row]+126}" font-family="Arial, sans-serif" font-size="29" font-weight="700" fill="#f1e7d5">${escape(lines[0])}</text><text x="60" y="${rowY[row]+165}" font-family="Arial, sans-serif" font-size="27" fill="#c2cbd7">${escape(lines[1])}</text>`
}).join('')}
${rowY.flatMap(y => colX.map(x => `<rect x="${x}" y="${y}" width="300" height="300" rx="12" fill="#0b1a2b" stroke="#25354a" stroke-width="2"/>`)).join('')}
<text x="60" y="1230" font-family="Arial, sans-serif" font-size="22" fill="#91a2b6">Typography review only. The eleven approved symbol images remain unchanged.</text>
</svg>`
const layers = [{ input: Buffer.from(backdrop), left: 0, top: 0 }]
for (let i = 0; i < glyphs.length; i++) layers.push({ input: glyphs[i].preview, left: colX[i%4], top: rowY[Math.floor(i/4)] })
const board = await sharp({ create: { width, height, channels: 4, background } }).composite(layers).png().toBuffer()
await writeFile(resolve(root, 'font-study-board.png'), board)
await sharp(board).resize(1224).png().toFile(resolve(root, 'font-study-board-1224.png'))

const strips = []
for (const family of sourceManifest.families) {
  const inputs = []
  for (let i=0;i<4;i++) inputs.push({ input: await sharp(await readFile(resolve(root, family.assets[i].file))).resize(96,96,{fit:'contain'}).png().toBuffer(), left:20+i*120, top:20 })
  const file = `${family.id}-96px-canvas-strip.png`
  await sharp({ create: { width:496, height:136, channels:4, background } }).composite(inputs).png().toFile(resolve(root,file))
  strips.push({ family: family.title, file, perGlyphCanvas:96, sourceCanvas:1254, cropApplied:false, baselineScale:96/1254 })
}

const records = glyphs.map(({preview,...item}) => item)
const verification = { checkedUtc:new Date().toISOString(), kind:'Native font study packaging only',
  approvedImagesChecked:11, allApprovedImagesMatchRevision2:true, approvedImages:approvedProof,
  glyphsChecked:12, familyCount:3, eachFamilyOrder:'AKQJ', sourceCanvas:[1254,1254],

  sourceFontFilesCopiedOrEmbedded:false, sourcePngsModified:false, imageGenerationUsed:false,
  noGameOrProductionChanges:true }
await writeFile(resolve(root,'font-study-verification.json'),JSON.stringify(verification,null,2)+'\n')
const manifest = { kind:'Three native font directions for owner review; no font selected or final fifteen-symbol board created',
  board:{full:'font-study-board.png',preview:'font-study-board-1224.png',width,height,background},
  nativeFontSourceManifest:'font-studies-manifest.json', fill:'#f1e7d5', effects:[],
  layout:'Three family rows and four letter columns. Entire source canvases uniformly scaled; supplied baselines and native proportions preserved.',
  records, strips, verification:'font-study-verification.json', approvedImagesUnchanged:11 }
