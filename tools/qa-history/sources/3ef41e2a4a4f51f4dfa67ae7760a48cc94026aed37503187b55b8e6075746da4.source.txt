import fs from 'node:fs'
import path from 'node:path'
import postcss from 'postcss'
import { createHash } from 'node:crypto'

const root = process.cwd()
const out = path.join(root, 'output/maintenance-2026-11-01/navbar-consistency')
const scope = JSON.parse(fs.readFileSync(path.join(out, 'css-owned-scope-before.json'), 'utf8').replace(/^\uFEFF/, ''))
const readRules = text => { const rules = []; postcss.parse(text).walkRules(rule => rules.push({ selector: rule.selector,
  declarations: rule.nodes.filter(node => node.type === 'decl').map(node => `${node.prop}:${node.value}${node.important ? '!important' : ''}`) })); return rules }
for (const entry of scope) {
  const before = fs.readFileSync(path.join(out, 'before', entry.path), 'utf8')
  const after = fs.readFileSync(path.join(root, entry.path), 'utf8')
  const oldRules = readRules(before), newRules = readRules(after)
  entry.afterSha256 = createHash('sha256').update(after).digest('hex')
  entry.ruleCounts = { before: oldRules.length, after: newRules.length }
  entry.changedRules = newRules.flatMap((rule, index) => JSON.stringify(rule) === JSON.stringify(oldRules[index]) ? [] : [{ before: oldRules[index], after: rule }])
  entry.declarationValuesPreserved = oldRules.length === newRules.length && oldRules.every((rule, index) => JSON.stringify(rule.declarations) === JSON.stringify(newRules[index].declarations))
}
fs.writeFileSync(path.join(out, 'css-owned-changes.json'), JSON.stringify(scope, null, 2))
console.log(JSON.stringify(scope.map(entry => ({ path: entry.path, changedRules: entry.changedRules.length, declarationValuesPreserved: entry.declarationValuesPreserved })), null, 2))
