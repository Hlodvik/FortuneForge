async function provenance(id, file, group) {
  if (group === 'retained') {
    const original = `fortuneforge.client/public/assets/slots/midnight-vault/v1/symbols/320/${basename(file)}`
    const bytes = await readFile(resolve(repo, original))
    return { method: 'Exact unchanged original V1 WebP copy', original, originalSha256: hash(bytes), prompt: null }
  }
  if (group === 'props') {
    const item = props.assets.find((asset) => asset.slug === id)
    if (!item) throw new Error(`Missing prop provenance: ${id}`)
    return { method: 'Built-in image_gen; original V1 ring style reference only', generatedSource: item.generatedSource,
      sourceManifest: 'props/manifest.json', promptFile: `props/${id}.prompt.txt`, prompt: await readFile(item.prompt, 'utf8') }
  }
  if (group === 'cards') {
    const item = cards.assets.find((asset) => asset.asset === id)
    if (!item) throw new Error(`Missing rank provenance: ${id}`)
    return { method: 'Built-in image_gen; original V1 ring style reference only', generatedSource: item.source,
      sourceManifest: 'cards/source-manifest.json', promptFile: 'cards/prompt-set.json', prompt: item.prompt }
  }
  const item = specials.assets.find((asset) => asset.name === id)
  if (!item) throw new Error(`Missing special provenance: ${id}`)
  return { method: 'Built-in image_gen; original V1 ring style reference only', generatedSource: item.generatedSource,
    sourceManifest: 'specials/manifest.json', promptFile: `specials/${id}.prompt.txt`, prompt: await readFile(resolve(root, 'specials', item.prompt), 'utf8') }
}

async function imageBounds(bytes) {
  const { data, info } = await sharp(bytes).ensureAlpha().raw().toBuffer({ resolveWithObject: true })
  let left = info.width, top = info.height, right = -1, bottom = -1, transparent = 0, min = 255, max = 0
  for (let y = 0; y < info.height; y++) {
    for (let x = 0; x < info.width; x++) {
      const alpha = data[(y * info.width + x) * 4 + 3]
      min = Math.min(min, alpha); max = Math.max(max, alpha)
      if (alpha === 0) transparent++
      if (alpha > 8) { left = Math.min(left, x); right = Math.max(right, x); top = Math.min(top, y); bottom = Math.max(bottom, y) }
    }
  }
  if (right < left || bottom < top || min !== 0 || max !== 255) throw new Error('Missing visible art or genuine transparency')
  return { width: info.width, height: info.height, alphaRange: [min, max], fullyTransparentPixels: transparent,
    visibleBoundsAtAlpha8: [left, top, right + 1, bottom + 1],
    extract: { left, top, width: right - left + 1, height: bottom - top + 1 } }
}

const assets = []
for (let index = 0; index < sourceList.length; index++) {
  const [id, label, file, group] = sourceList[index]
  const originalBytes = await readFile(resolve(root, file))
  const origin = await provenance(id, file, group)
  const bytesHash = hash(originalBytes)
  if (group === 'retained' && bytesHash !== origin.originalSha256) throw new Error(`Retained image changed: ${file}`)
  if (group !== 'retained' && bytesHash !== hash(await readFile(origin.generatedSource))) throw new Error(`Generated source changed: ${file}`)
  const bounds = await imageBounds(originalBytes)
  // Only board layout is normalized: trim invisible margins in memory and
  // contain each subject in a common 300px square. No individual asset is saved.
  const preview = await sharp(originalBytes).extract(bounds.extract).resize(300, 300, { fit: 'inside' }).png().toBuffer({ resolveWithObject: true })
  assets.push({ index: index + 1, id, label, file, group, byteCount: originalBytes.length,
    sha256: bytesHash, ...origin, ...bounds, sourceUnmodified: true, preview })
}
