def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def generated_path(item):
    return Path(item.get('generated_path') or item.get('sourceMasterPath') or
                item['output_hint'].split(' as ', 1)[1].split(' by default.', 1)[0])

def font(size):

        if Path(name).exists():
            return ImageFont.truetype(name, size)
    return ImageFont.load_default()

def framed(path, size):
    im = Image.open(path).convert('RGBA')
    # Ignore nearly invisible generation specks for framing only, preserve edge alpha.
    bounds = im.getchannel('A').point(lambda a: 255 if a > 8 else 0).getbbox()
    if not bounds:
        raise ValueError(f'No visible alpha: {path}')
    im = ImageOps.contain(im.crop(bounds), (round(size * .88), round(size * .88)), Image.Resampling.LANCZOS)
    result = Image.new('RGBA', (size, size), (0, 0, 0, 0))
    result.alpha_composite(im, ((size - im.width) // 2, (size - im.height) // 2))
    return result

def save_webp(im, dest, budget=None, start_quality=88):
    dest.parent.mkdir(parents=True, exist_ok=True)
    quality = start_quality
    while True:
        buf = io.BytesIO()
        im.save(buf, 'WEBP', quality=quality, method=6, exact=True)
        data = buf.getvalue()
        if not budget or len(data) <= budget or quality <= 80:
            break
        quality -= 2
    dest.write_bytes(data)
    return quality

def record(slug, role, src, dest, quality, prompt_file, kind):
    with Image.open(dest) as im:
        alpha = im.getchannel('A').getextrema() if 'A' in im.getbands() else None
        row = dict(slug=slug, role=role, kind=kind, path=dest.relative_to(OUT).as_posix(),
                   width=im.width, height=im.height, bytes=dest.stat().st_size,
                   format=im.format, alpha_min=alpha[0] if alpha else '', alpha_max=alpha[1] if alpha else '',
                   webp_quality=quality, sha256=sha(dest), source_master=str(src),
                   source_sha256=sha(src), prompt_file=prompt_file)
        rows.append(row)

def reel_preview(width):
    height=round(width/1.56)
    bg=ImageOps.fit(Image.open(OUT/'presentation/midnight-vault-backdrop-1672x941.webp').convert('RGBA'),(width,height),method=Image.Resampling.LANCZOS)
    draw=ImageDraw.Draw(bg)
    pad=round(width*.02); gap=round(width*.008)
    cw=(width-2*pad-4*gap)/5; ch=(height-2*pad-3*gap)/4
    pixels=round(min(cw,ch)*.82)
    order=list(ROLES)+['ruby-ring','master-key','vault-bonus']
    for i,slug in enumerate(order):
        c=i%5; r=i//5
        x=round(pad+c*(cw+gap)); y=round(pad+r*(ch+gap))
        # A restrained transparent well approximates the dark-well review concept.
        well=Image.new('RGBA',bg.size,(0,0,0,0)); wd=ImageDraw.Draw(well)
        wd.rounded_rectangle((x,y,x+round(cw),y+round(ch)),radius=max(2,round(width*.005)),fill=(3,10,20,145),outline=(139,119,79,85))
        bg=Image.alpha_composite(bg,well)
        art=symbol(slug,pixels)
        bg.alpha_composite(art,(round(x+(cw-pixels)/2),round(y+(ch-pixels)/2)))
    return bg.convert('RGB'),pixels
