async function provenance(id, file, group) {
  const previous = previousManifest.assets.find((asset) => asset.id === id)
  if (!previous) throw new Error(`Missing previous symbol identity: ${id}`)
  if (group === 'accepted') return { method: 'Accepted previous art, reused without any source modification',
    previousFile: `../${previous.file}`, previousSha256: previous.sha256,
    previousProvenance: { method: previous.method, generatedSource: previous.generatedSource ?? null,
      original: previous.original ?? null, prompt: previous.prompt ?? null, promptFile: previous.promptFile ? `../${previous.promptFile}` : null } }
  const folder = dirname(resolve(root, file))
  const records = []
  for (const name of (await readdir(folder)).filter((name) => /\.(?:json|txt)$/i.test(name)).sort()) {
    const content = await readFile(resolve(folder, name), 'utf8')
    records.push({ file: `${dirname(file).replaceAll('\\', '/')}/${name}`, sha256: hash(Buffer.from(content)),
      content: name.endsWith('.json') ? JSON.parse(content) : content })
  }
  if (!records.length) throw new Error(`Missing exact prompt/source provenance for revised symbol: ${id}`)
  return { method: 'Built-in image_gen revision; exact source/prompt records attached', previousFile: `../${previous.file}`,
    previousSha256: previous.sha256, revisionProvenance: records }
}

async function imageBounds(bytes) {
  const { data, info } = await sharp(bytes).ensureAlpha().raw().toBuffer({ resolveWithObject: true })
  let left = info.width, top = info.height, right = -1, bottom = -1, transparent = 0, min = 255, max = 0
  for (let y = 0; y < info.height; y++) {
    for (let x = 0; x < info.width; x++) {
      const alpha = data[(y * info.width + x) * 4 + 3]
      min = Math.min(min, alpha); max = Math.max(max, alpha)
      if (alpha === 0) transparent++
      if (alpha > 8) { left = Math.min(left, x); right = Math.max(right, x); top = Math.min(top, y); bottom = Math.max(bottom, y) }
    }
  }
  if (right < left || bottom < top || min !== 0 || max !== 255) throw new Error('Missing visible art or genuine transparency')
  return { width: info.width, height: info.height, alphaRange: [min, max], fullyTransparentPixels: transparent,
    visibleBoundsAtAlpha8: [left, top, right + 1, bottom + 1],
    extract: { left, top, width: right - left + 1, height: bottom - top + 1 } }
}

  const preview = await sharp(originalBytes).extract(bounds.extract).resize(300, 300, { fit: 'inside' }).png().toBuffer({ resolveWithObject: true })
  assets.push({ index: index + 1, id, label, file, group, byteCount: originalBytes.length,
    sha256: bytesHash, ...origin, ...bounds, sourceUnmodified: true, preview })
}
